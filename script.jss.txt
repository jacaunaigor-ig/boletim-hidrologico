function esc(s) {
    return String(s ?? "").replace(/[&<>"']/g, m => ({
        '&': '&amp;',
        '<': '&lt;',
        '>': '&gt;',
        '"': '&quot;',
        "'": '&#039;'
    } [m]));
}

function cor(st) {
    return st === "ALTO" ? "#f97316" : st === "MODERADO" ? "#f59e0b" : "#10b981";
}

// Ajuste do nome da classe visual (Baixo ao invés de Fraco)
function pill(st) {
    return st === "ALTO" ? "p-alto" : st === "MODERADO" ? "p-moderado" : st === "NORMAL" ? "p-baixo" : "p-none";
}

function getStatusAtivo(e) {
    const tipo = document.getElementById("filtro-tipo").value;
    if (tipo === "vazante") return e.status_vazante || "NORMAL";
    if (tipo === "enchente") return e.status_enchente || "NORMAL";
    return e.status || "NORMAL";
}

function filtrados() {
    const calha = document.getElementById("calha").value;
    const status = document.getElementById("status").value;
    const busca = document.getElementById("busca").value.toLowerCase().trim();

    return DADOS.filter(e => {
        const st = getStatusAtivo(e);
        const isSL = e.cota === null;
        const matchStatus = status === "Todos" || (status === "SL" ? isSL : st === status);
        const matchCalha = calha === "Todas" || e.calha === calha;
        const matchBusca = !busca || e.municipio.toLowerCase().includes(busca) || e.ana.toLowerCase().includes(busca);
        return matchStatus && matchCalha && matchBusca;
    });
}

function addMarkers() {
    // Limpa marcadores antigos
    if (markerCluster) {
        map.removeLayer(markerCluster);
        markerCluster = null;
    }
    markers.forEach(m => map.removeLayer(m));
    markers = [];

    const listaFiltrada = filtrados();
    if (listaFiltrada.length === 0) return;

    // Cria cluster group
    markerCluster = L.markerClusterGroup({
        spiderfyOnMaxZoom: true,
        showCoverageOnHover: false,
        zoomToBoundsOnClick: true,
        maxClusterRadius: 35
    });

    listaFiltrada.forEach(e => {
        if (e.lat == null || e.lng == null) return;
        const st = getStatusAtivo(e);
        const markerClass = st === "ALTO" ? "high" : st === "MODERADO" ? "moderate" : "normal";
        const icon = L.divIcon({
            className: "modern-marker-wrap",
            html: `<div class="modern-marker ${markerClass}" style="background:${cor(st)}"></div>`,
            iconSize: [18, 18],
            iconAnchor: [9, 9],
            popupAnchor: [0, -8]
        });
        const m = L.marker([e.lat, e.lng], { icon, title: e.municipio, keyboard: true });
        m.bindTooltip(`<b>${esc(e.municipio)}</b>`, {
            direction: "top",
            offset: [0, -8],
            opacity: .9,
            className: 'custom-tooltip'
        });
        m.bindPopup(popupHTML(e), { maxWidth: 280, className: "modern-popup" });
        m.on("popupopen", () => {
            const cv = m.getPopup().getElement().querySelector("canvas");
            if (!cv) return;
            if (chartAtual) chartAtual.destroy();
            chartAtual = new Chart(cv, {
                type: "line",
                data: {
                    labels: DIAS,
                    datasets: [{
                        data: e.cotas,
                        borderWidth: 2,
                        pointRadius: 2.5,
                        pointBackgroundColor: cor(st),
                        borderColor: cor(st),
                        backgroundColor: cor(st) + "22",
                        fill: true,
                        spanGaps: false,
                        tension: .35
                    }]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    plugins: { legend: { display: false } },
                    scales: {
                        x: {
                            ticks: { font: { size: 7 }, color: '#64748b' },
                            grid: { display: false }
                        },
                        y: {
                            ticks: { font: { size: 7 }, color: '#64748b' },
                            grid: { color: "rgba(148,163,184,.15)" }
                        }
                    }
                }
            });
        });
        markerCluster.addLayer(m);
        markers.push(m);
    });

    map.addLayer(markerCluster);
}

// ---------- GEOLOCALIZAÇÃO (Estação Próxima) ----------
function encontrarMaisProximo() {
    if (!navigator.geolocation) {
        showToast("Geolocalização não é suportada pelo seu navegador.", "error");
        return;
    }

    const btn = document.querySelector('.btn-geo');
    const textoOriginal = btn.innerHTML;
    btn.innerHTML = '⏳ <span>Buscando...</span>';
    btn.disabled = true;

    // Função Haversine para cálculo preciso de distância em km
    function calcularDistancia(lat1, lon1, lat2, lon2) {
        const R = 6371; // Raio da Terra em km
        const dLat = (lat2 - lat1) * (Math.PI / 180);
        const dLon = (lon2 - lon1) * (Math.PI / 180);
        const a = Math.sin(dLat / 2) * Math.sin(dLat / 2) +
            Math.cos(lat1 * (Math.PI / 180)) * Math.cos(lat2 * (Math.PI / 180)) *
            Math.sin(dLon / 2) * Math.sin(dLon / 2);
        const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
        return R * c; // Distância em km
    }

    navigator.geolocation.getCurrentPosition(
        // SUCESSO
        position => {
            const userLat = position.coords.latitude;
            const userLng = position.coords.longitude;

            let maisProximo = null;
            let menorDistancia = Infinity;

            DADOS.forEach(e => {
                if (e.lat != null && e.lng != null) {
                    const dist = calcularDistancia(userLat, userLng, e.lat, e.lng);
                    if (dist < menorDistancia) {
                        menorDistancia = dist;
                        maisProximo = e;
                    }
                }
            });

            if (maisProximo) {
                if (window.innerWidth <= 768 && listaVisivel) {
                    alternarVisibilidadeLista();
                }
                map.setView([maisProximo.lat, maisProximo.lng], 9);
                if (markerCluster) {
                    markerCluster.eachLayer(layer => {
                        const ll = layer.getLatLng();
                        if (ll.lat === maisProximo.lat && ll.lng === maisProximo.lng) {
                            layer.openPopup();
                        }
                    });
                }
                showToast(`📍 Estação mais próxima: ${maisProximo.municipio} (${menorDistancia.toFixed(1)} km)`, "success");
            } else {
                showToast("Nenhuma estação com coordenadas encontrada.", "error");
            }

            btn.innerHTML = textoOriginal;
            btn.disabled = false;
        },
        // ERRO
        error => {
            console.warn("Erro de geolocalização:", error.message);
            if (error.code === 1) {
                showToast("Permissão de localização negada. Ative nas configurações do navegador.", "error");
            } else if (error.code === 2) {
                showToast("Localização indisponível no momento.", "error");
            } else if (error.code === 3) {
                showToast("Tempo limite excedido. Tente novamente.", "error");
            } else {
                showToast("Erro ao obter localização.", "error");
            }
            btn.innerHTML = textoOriginal;
            btn.disabled = false;
        },
        {
            enableHighAccuracy: true,
            timeout: 10000,
            maximumAge: 60000
        }
    );
}

function render() {
    if (debounceTimer) clearTimeout(debounceTimer);
    debounceTimer = setTimeout(() => {
        const listaFiltrada = filtrados();
        atualizarKPIs();
        addMarkers();

        const legendContent = document.getElementById("map-legend-content");
        if (legendContent) atualizarConteudoLegenda(legendContent);

        miniCharts.forEach(chart => chart.destroy());
        miniCharts = [];

        const lista = document.getElementById("list");
        lista.innerHTML = "";

        if (listaFiltrada.length === 0) {
            lista.innerHTML = `<div style="padding:20px; text-align:center; font-size:13px; color:var(--text-muted);">
        <div style="font-size:32px; margin-bottom:8px;">🔍</div>
        <b>Nenhum resultado</b>
      </div>`;
            return;
        }

        listaFiltrada.forEach((e, idx) => {
            const st = getStatusAtivo(e);
            const div = document.createElement("div");
            div.className = "item";
            const miniId = "mini_" + idx;
            div.innerHTML = `
        <div>
          <b>${esc(e.municipio)}</b><br>
          <small>${esc(e.calha)} · ${e.cota != null ? e.cota.toFixed(2) + "m" : "SL"}</small><br>

        </div>
        <div class="item-mini-chart"><canvas id="${miniId}"></canvas></div>
      `;
            div.onclick = () => {
                if (window.innerWidth <= 768) alternarVisibilidadeLista();
                if (e.lat != null && e.lng != null) {
                    map.setView([e.lat, e.lng], 9);
                    if (markerCluster) {
                        markerCluster.eachLayer(layer => {
                            const ll = layer.getLatLng();
                            if (ll.lat === e.lat && ll.lng === e.lng) {
                                layer.openPopup();
                            }
                        });
                    }
                }
            };
            lista.appendChild(div);

            setTimeout(() => {
                const canvasEl = document.getElementById(miniId);
                if (canvasEl) {
                    const newChart = new Chart(canvasEl, {
                        type: "line",
                        data: {
                            labels: DIAS,
                            datasets: [{

                }
            }, 0);
        });
    }, 100);

function configurarGestosSwipe() {
    const aside = document.getElementById("panelAside");
    if (!aside) return;
    aside.addEventListener('touchstart', e => {
        touchstartY = e.changedTouches[0].screenY;
    }, { passive: true });
    aside.addEventListener('touchend', e => {
        touchendY = e.changedTouches[0].screenY;
        if (touchendY - touchstartY > 100 && window.innerWidth <= 768 && listaVisivel) {
            alternarVisibilidadeLista();
        }
    }, { passive: true });
}

// ---------- EXPORTAÇÃO ----------
function exportarCSV() {
    const listaFiltrada = filtrados();
    let csv = "Municipio;Calha;ANA;Cota(m);Tendencia;Status\n";
    listaFiltrada.forEach(e => {
        csv += `"${e.municipio}";"${e.calha}";"${e.ana}";"${e.cota !== null ? e.cota : 'SL'}";"${e.tendencia}";"${getStatusAtivo(e)}"\n`;
    });

    const blob = new Blob(["\uFEFF" + csv], { type: 'text/csv;charset=utf-8;' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `boletim_hidrologico_${new Date().toISOString().slice(0,10)}.csv`;
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
    showToast("📥 CSV exportado", "success");
}
